Overriding - 
	class Parent{
		virtual int Addition(int i, int j){
			return i + j;
		}
	}

	class Child : Parent{
		override int Addition(int i, int j){
			return i + j + 1000;
		}
	}
Parent p1=new Parent();
p1.Addition(100,200); //300

Child p2=new Child();
p2.Addition(100,200); //1300

Parent p3=new Child();
p3.Addition(100,200); //1300


Shadowing
	class Parent{
		int Addition(int i, int j){
			return i + j;
		}
	}

	class Child : Parent{
		new int Addition(int i, int j){
			return i + j + 1000;
		}
	}

Parent p3=new Child();
p3.Addition(100,200); //300
